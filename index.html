const SHEET_VOTES = 'Votes'; const SHEET_QUESTIONS = 'Questions'; const SHEET_CONFIG = 'Config'; const TOTAL_PLAYERS = 9; const ADMIN_USER = "Trinh"; function doGet(e) { const action = e.parameter.action; const ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Tải câu hỏi & Vòng hiện tại if (action === 'getQuestions') { const qSheet = ss.getSheetByName(SHEET_QUESTIONS); if (!qSheet) return createJsonResponse({ questions: [] }); const data = qSheet.getDataRange().getValues(); const questions = []; for (let i = 1; i < data.length; i++) { if (!data[i][0]) continue; questions.push({ id: String(data[i][0]).trim(), title: String(data[i][1]).trim(), type: String(data[i][2]).trim(), description: String(data[i][3]).trim(), rawOptions: String(data[i][4] || '').trim() }); } const cfgSheet = getOrCreateSheet(ss, SHEET_CONFIG); const activeRound = parseInt(cfgSheet.getRange('A1').getValue() || "0"); return createJsonResponse({ questions: questions, activeRound: activeRound }); } // 2. Lấy trạng thái voter if (action === 'getVoterState') { const voter = e.parameter.voter; const vSheet = ss.getSheetByName(SHEET_VOTES); const data = vSheet ? vSheet.getDataRange().getValues() : []; const submittedRounds = {}; const votedUsersByRound = { 0: new Set(), 1: new Set(), 2: new Set() }; for (let i = 1; i < data.length; i++) { const vName = String(data[i][0]).trim(); const qId = String(data[i][1]).trim(); const roundIdx = parseInt(data[i][3] || "0"); if (vName) votedUsersByRound[roundIdx]?.add(vName); if (vName === voter) { submittedRounds[qId] = String(data[i][2]).split(' | '); } } const cfgSheet = getOrCreateSheet(ss, SHEET_CONFIG); const activeRound = parseInt(cfgSheet.getRange('A1').getValue() || "0"); return createJsonResponse({ submittedRounds: submittedRounds, votedCounts: { 0: votedUsersByRound[0] ? votedUsersByRound[0].size : 0, 1: votedUsersByRound[1] ? votedUsersByRound[1].size : 0, 2: votedUsersByRound[2] ? votedUsersByRound[2].size : 0 }, activeRound: activeRound }); } // 3. Lấy bảng kết quả (Chỉ mở cho Trinh) if (action === 'getResults') { const voter = e.parameter.voter; const vSheet = ss.getSheetByName(SHEET_VOTES); const data = vSheet ? vSheet.getDataRange().getValues() : []; if (voter !== ADMIN_USER) { return createJsonResponse({ status: 'RESTRICTED' }); } const tally = {}; for (let i = 1; i < data.length; i++) { const qId = String(data[i][1]).trim(); const ans = String(data[i][2]).trim(); if (qId && ans) { if (!tally[qId]) tally[qId] = {}; const choices = ans.split(' | '); choices.forEach(c => { tally[qId][c] = (tally[qId][c] || 0) + 1; }); } } return createJsonResponse({ status: 'UNLOCKED', results: tally }); } } function doPost(e) { try { const body = JSON.parse(e.postData.contents); const action = body.action; const ss = SpreadsheetApp.getActiveSpreadsheet(); // Reset Game (Chỉ Host Trinh được bấm) if (action === 'resetGame') { const vSheet = ss.getSheetByName(SHEET_VOTES); if (vSheet && vSheet.getLastRow() > 1) { vSheet.getRange(2, 1, vSheet.getLastRow() - 1, vSheet.getLastColumn()).clearContent(); } const cfgSheet = getOrCreateSheet(ss, SHEET_CONFIG); cfgSheet.getRange('A1').setValue(0); return createJsonResponse({ success: true, message: "Game reset successfully!" }); } // Chuyển vòng thi if (action === 'setRound') { if (body.voter !== ADMIN_USER) return createJsonResponse({ success: false, message: 'Unauthorized' }); const cfgSheet = getOrCreateSheet(ss, SHEET_CONFIG); cfgSheet.getRange('A1').setValue(body.round); return createJsonResponse({ success: true, activeRound: body.round }); } // Ghi nhận phiếu vote const voter = body.voter; const qId = body.qId; const answer = body.answer; const roundIdx = body.roundIndex; const vSheet = getOrCreateSheet(ss, SHEET_VOTES); const ansVal = Array.isArray(answer) ? answer.join(' | ') : answer; vSheet.appendRow([voter, qId, ansVal, roundIdx, new Date()]); return createJsonResponse({ success: true }); } catch(err) { return createJsonResponse({ success: false, error: err.toString() }); } } function getOrCreateSheet(ss, name) { let sheet = ss.getSheetByName(name); if (!sheet) sheet = ss.insertSheet(name); return sheet; } function createJsonResponse(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }
